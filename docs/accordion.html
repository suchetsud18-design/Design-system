Usage — shadcn/ui

# Install via shadcn/ui CLI
npx shadcn@latest add accordion
import {
  Accordion, AccordionContent,
  AccordionItem, AccordionTrigger,
} from "@/components/ui/accordion"

export function AccordionDemo() {
  return (
    <Accordion type="single" collapsible className="w-full">
      <AccordionItem value="item-1">
        <AccordionTrigger>Product Information</AccordionTrigger>
        <AccordionContent>
          Our flagship product combines cutting-edge technology.
        </AccordionContent>
      </AccordionItem>
      <AccordionItem value="item-2">
        <AccordionTrigger>Shipping Details</AccordionTrigger>
        <AccordionContent>Standard delivery 3–5 days.</AccordionContent>
      </AccordionItem>
    </Accordion>
  )
}

Props

ComponentPropTypeDefaultDescription
Accordiontype"single" | "multiple"—Single or multiple open items.
AccordioncollapsiblebooleanfalseAllow closing when type="single".
AccordiondefaultValuestring | string[]—Default open item(s) — uncontrolled.
Accordionvaluestring | string[]—Controlled open item(s).
AccordionItemvaluestring—Unique identifier for this item.
AccordionItemdisabledbooleanfalsePrevents interaction.
AccordionTriggerasChildbooleanfalseMerge props onto child (Radix slot).
